<resultMap id="membershipResultMap" type="Membership">
    <result property="employerNo" column="EMPLOYER_NO"/>
    <result property="membershipNo" column="MEMBERSHIP_NO"/>
    <result property="membershipType" column="MEMBERSHIP_TYPE"/>
    <result property="membershipStartDate" column="MEMBERSHIP_START_DATE"/>
    <result property="membershipEndDate" column="MEMBERSHIP_END_DATE"/>
    <result property="membershipExpirationDate" column="MEMBERSHIP_EXPIRATION_DATE"/>
    <result property="remainingDays" column="REMAINING_DAYS"/>
    <result property="membershipName" column="MEMBERSHIP_NAME"/>
    <result property="membershipDelFl" column="MEMBERSHIP_DEL_FL"/>
</resultMap>


<select id="selectMembershipDetails" resultMap="membershipResultMap">
    SELECT 
        A.EMPLOYER_NO AS EMPLOYER_NO,
        B.MEMBERSHIP_NO AS MEMBERSHIP_NO,
        B.MEMBERSHIP_TYPE AS MEMBERSHIP_TYPE,
        B.MEMBERSHIP_START_DATE AS MEMBERSHIP_START_DATE,
        B.MEMBERSHIP_END_DATE AS MEMBERSHIP_END_DATE,
        B.MEMBERSHIP_EXPIRATION_DATE AS MEMBERSHIP_EXPIRATION_DATE,
        (TRUNC(B.MEMBERSHIP_END_DATE) - TRUNC(B.MEMBERSHIP_START_DATE)) AS REMAINING_DAYS,
        (SELECT C.MEMBERSHIP_NAME
            FROM MEMBERSHIP_LEVEL C
            WHERE C.MEMBERSHIP_NO = B.MEMBERSHIP_TYPE) AS MEMBERSHIP_NAME
    FROM EMPLOYER A
    JOIN MEMBERSHIP B ON A.EMPLOYER_NO = B.EMPLOYER_NO
    WHERE A.EMPLOYER_NO = #{employerNo}
    AND B.MEMBERSHIP_DEL_FL = 'N'
</select>
